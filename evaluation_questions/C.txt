def export_to_netconf(self, chromosome, out_path, filename_prefix="tsn_schedule"):
    """
    Exports a final TSN schedule (chromosome) to a NETCONF/YANG configuration file for deployment on real switches.
    
    Args:
        chromosome: 2D numpy array (n_ports, n_streams) representing the final schedule.
        out_path: Directory path where NETCONF files will be written.
        filename_prefix: Prefix for the generated filenames.
    
    Returns:
        dict: Mapping of switch names to their respective NETCONF file paths.
    """
    import os
    import numpy as np
    from lxml import etree
    
    if not os.path.exists(out_path):
        os.makedirs(out_path)
    
    # Validate chromosome shape
    if chromosome.shape != (self.n_ports, self.n_streams):
        raise ValueError(f"Chromosome shape {chromosome.shape} does not match expected ({self.n_ports}, {self.n_streams})")
    
    # Group ports by switch
    switch_ports = {}
    for port_idx in range(self.n_ports):
        node_id = self.port_id_to_node[port_idx + 1]  # port_id_to_node is 1-based
        if node_id not in switch_ports:
            switch_ports[node_id] = []
        switch_ports[node_id].append(port_idx)
    
    # Generate NETCONF config for each switch
    file_paths = {}
    
    for node_id, ports in switch_ports.items():
        # Determine switch name from nodes dict
        switch_name = None
        for name, (nid, _) in self.nodes.items():
            if nid == node_id:
                switch_name = name
                break
        
        if switch_name is None:
            continue
        
        # Build NETCONF configuration
        root = etree.Element("config")
        root.set("xmlns", "urn:ietf:params:xml:ns:netconf:base:1.0")
        
        # Add TSN scheduling configuration
        tsn_config = etree.SubElement(root, "tsn-scheduling")
        tsn_config.set("xmlns", "urn:ietf:params:xml:ns:yang:ietf-tsn")
        
        # Add schedule details
        schedule = etree.SubElement(tsn_config, "schedule")
        schedule.set("type", self.scheduling_type.name)
        
        # Add port configurations
        for port_idx in ports:
            port_config = etree.SubElement(schedule, "port")
            port_config.set("id", str(port_idx + 1))
            
            # Add stream configurations for this port
            for stream_idx in range(self.n_streams):
                value = chromosome[port_idx, stream_idx]
                if value > 0:  # Active stream
                    stream_config = etree.SubElement(port_config, "stream")
                    stream_config.set("id", str(stream_idx + 1))
                    
                    # Add scheduling parameters based on type
                    if self.scheduling_type.value == "PRIORITY":
                        priority = etree.SubElement(stream_config, "priority")
                        priority.text = str(int(value))
                    elif self.scheduling_type.value == "START_TIME":
                        start_time = etree.SubElement(stream_config, "start-time")
                        start_time.text = str(int(value))
                        
                        # Add window length
                        window_length = etree.SubElement(stream_config, "window-length")
                        window_length.text = str(self.windLengths[port_idx, stream_idx])
                        
                        # Add interval
                        interval = etree.SubElement(stream_config, "interval")
                        interval.text = str(self.stream_intervals[stream_idx])
        
        # Write to file
        filename = f"{filename_prefix}_{switch_name}.xml"
        filepath = os.path.join(out_path, filename)
        
        tree = etree.ElementTree(root)
        tree.write(filepath, pretty_print=True, xml_declaration=True, encoding="UTF-8")
        
        file_paths[switch_name] = filepath
    
    return file_paths

def generate_gate_control_lists(chromosome, problem, macro_cycle=1000000):
    """
    Generates IEEE 802.1Qbv Gate Control Lists (GCLs) for each switch port from a computed schedule chromosome.
    
    Args:
        chromosome: 2D numpy array (n_ports, n_streams) where chromosome[p, s] is start time (0 = inactive)
        problem: Problem object containing topology and stream configuration
        macro_cycle: Scheduling period in nanoseconds (default 1e6 ns = 1ms)
    
    Returns:
        dict: {port_id: [gate_entry1, gate_entry2, ...]}
            where each gate_entry = {'start_time': int, 'duration': int, 'stream_id': int, 'gate_state': 'open'}
    
    Assumptions:
        - Port IDs are 1-based integers
        - Stream IDs are 0-based integers
        - Gate state is always 'open' for active streams (closed gaps are implicit)
        - Entries are sorted by start_time within each port
    """
    import numpy as np
    
    # Validate chromosome dimensions
    if not isinstance(chromosome, np.ndarray):
        raise ValueError("chromosome must be a numpy array")
    if chromosome.ndim != 2:
        raise ValueError("chromosome must be a 2D array")
    if chromosome.dtype not in [np.int32, np.int64, np.int8, np.int16]:
        raise ValueError("chromosome must contain integers")
    
    n_ports, n_streams = chromosome.shape
    
    # Validate problem data completeness
    if not isinstance(problem, dict):
        raise KeyError("problem must be a dictionary")
    if 'topology' not in problem:
        raise KeyError("problem missing 'topology' key")
    if 'streams' not in problem:
        raise KeyError("problem missing 'streams' key")
    
    # Precompute stream durations (transmission time per port)
    port_speeds = np.zeros(n_ports, dtype=float)
    for p_idx in range(n_ports):
        pid = p_idx + 1
        if pid not in problem['topology'].get('ports', {}):
            raise KeyError(f"Port {pid} not found in topology")
        pdata = problem['topology']['ports'][pid]
        port_speeds[p_idx] = pdata["speed"]
    
    gcl = {}
    
    # Build GCL per port
    for p_idx in range(n_ports):
        pid = p_idx + 1
        port_entries = []
        
        for s_idx in range(n_streams):
            start_time = chromosome[p_idx, s_idx]
            if start_time == 0:
                continue  # Skip inactive streams
            
            # Compute transmission duration
            if s_idx >= len(problem['streams']):
                raise KeyError(f"Stream index {s_idx} out of bounds")
            stream_data = problem['streams'][s_idx]
            frame_size = stream_data.get('frame_size', 1500)  # Default to 1500 bytes if not specified
            
            duration = frame_size * 8 / max(port_speeds[p_idx] * 1e6, 1e-9)  # Convert to ns (bits to ns)
            
            port_entries.append({
                'start_time': int(start_time),
                'duration': int(duration),
                'stream_id': s_idx,
                'gate_state': 'open'
            })
        
        # Sort by start time
        port_entries.sort(key=lambda x: x['start_time'])
        gcl[pid] = port_entries
    
    return gcl

def _per_stream_jitter(chromosome):
    """
    Computes the per-stream jitter (difference between maximum and minimum transmission start times across all active ports) for a given schedule chromosome.
    
    Args:
        chromosome: 2D numpy array of shape (n_ports, n_streams) where chromosome[p, s] represents the transmission start time (0 if inactive)
    
    Returns:
        1D numpy array of shape (n_streams,) containing the jitter value for each stream
    """
    import numpy as np
    
    active = chromosome > 0
    
    # Replace inactive entries so they don't affect min/max
    start_min = np.where(active, chromosome, np.inf)
    start_max = np.where(active, chromosome, -np.inf)
    
    # Per-stream earliest and latest transmission
    t_min = start_min.min(axis=0)
    t_max = start_max.max(axis=0)
    
    # Compute jitter as the difference between max and min start times
    jitter = t_max - t_min
    
    # Streams with no activity → 0 jitter
    jitter = np.where(np.isfinite(t_min), jitter, 0)
    
    return jitter

def mutation(chromosome, mutation_rate=0.01, mutation_magnitude=1.0, bounds=(0, 100)):
    """Applies random mutation to a chromosome."""
    import numpy as np
    
    if mutation_rate < 0 or mutation_rate > 1:
        raise ValueError("mutation_rate must be between 0 and 1")
    
    mutated = chromosome.copy()
    
    # Create a mask for mutation
    mask = np.random.random(chromosome.shape) < mutation_rate
    
    # Generate random perturbations
    perturbations = np.random.uniform(-mutation_magnitude, mutation_magnitude, chromosome.shape)
    
    # Apply perturbations
    mutated[mask] += perturbations[mask]
    
    # Clamp to bounds
    mutated = np.clip(mutated, bounds[0], bounds[1])
    
    return mutated

def check_link_capacity_violations(self, chromosome, macro_cycle=None):
    """
    Evaluates a candidate schedule to check for link capacity violations.
    
    Calculates the total transmission time required on each port based on the 
    chromosome assignments and compares it against the macrocycle duration.
    
    Args:
        chromosome: 2D numpy array (n_ports, n_streams) where values > 0 indicate active assignments.
        macro_cycle: Optional override for the macrocycle duration. If None, uses self.macro_cycle.
    
    Returns:
        A boolean numpy array of length n_ports. True indicates the port exceeds capacity (violation),
        False indicates the port is within capacity limits.
    """
    if macro_cycle is None:
        macro_cycle = self.macro_cycle
    
    if macro_cycle <= 0:
        raise ValueError("Macro cycle must be positive.")
    
    # Ensure chromosome is a numpy array
    chromosome = np.asarray(chromosome)
    
    if chromosome.ndim != 2 or chromosome.shape[0] != self.n_ports:
        raise ValueError(f"Chromosome must be a 2D array with shape ({self.n_ports}, n_streams).")
    
    # Calculate total load per port: sum of windLengths for all active streams on each port
    # windLengths shape: (n_ports, n_streams)
    # active mask: (n_ports, n_streams)
    active_mask = chromosome > 0
    
    # Total time required on each port
    total_load_per_port = np.sum(self.windLengths * active_mask, axis=1)
    
    # Check if total load exceeds macro_cycle
    # Violation occurs if load > macro_cycle
    violations = total_load_per_port > macro_cycle
    
    return violations
